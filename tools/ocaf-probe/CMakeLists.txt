cmake_minimum_required(VERSION 3.28)
project(VoelundOcafProbe LANGUAGES CXX)
include(CTest)
find_package(OpenCASCADE CONFIG REQUIRED)
add_executable(voelund_ocaf_probe main.cpp Fixture.cpp)
add_executable(voelund_identity_probe Identity.cpp Fixture.cpp ../../src/document/DocumentId.cpp)
target_include_directories(voelund_identity_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
foreach(probe_target IN ITEMS voelund_ocaf_probe voelund_identity_probe)
    target_compile_features(${probe_target} PRIVATE cxx_std_20)
    target_include_directories(${probe_target} SYSTEM PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
    target_link_libraries(${probe_target} PRIVATE TKBin TKCAF TKPrim)
    if(MSVC)
        target_compile_options(${probe_target} PRIVATE /W4 /WX /utf-8 /FS)
        target_link_options(${probe_target} PRIVATE /INCREMENTAL:NO)
        add_custom_command(TARGET ${probe_target} POST_BUILD
            COMMAND "${CMAKE_COMMAND}" "-DPROBE_EXE=$<TARGET_FILE:${probe_target}>"
                "-DOCCT_BIN=${OpenCASCADE_BINARY_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/DeployRuntime.cmake"
            VERBATIM)
    else()
        target_compile_options(${probe_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
add_test(NAME ocaf.create COMMAND voelund_ocaf_probe create "${CMAKE_CURRENT_BINARY_DIR}/fixtures/local.volund")
set_tests_properties(ocaf.create PROPERTIES FIXTURES_SETUP ocaf_file)
add_test(NAME ocaf.read_edit COMMAND voelund_ocaf_probe read "${CMAKE_CURRENT_BINARY_DIR}/fixtures/local.volund" "${CMAKE_CURRENT_BINARY_DIR}/fixtures/rewritten.volund")
set_tests_properties(ocaf.read_edit PROPERTIES FIXTURES_REQUIRED ocaf_file FIXTURES_SETUP ocaf_rewritten)
add_test(NAME ocaf.reread COMMAND voelund_ocaf_probe read "${CMAKE_CURRENT_BINARY_DIR}/fixtures/rewritten.volund")
set_tests_properties(ocaf.reread PROPERTIES FIXTURES_REQUIRED ocaf_rewritten)
add_test(NAME ocaf.negative COMMAND voelund_ocaf_probe negative "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
set_tests_properties(ocaf.create ocaf.read_edit ocaf.reread ocaf.negative PROPERTIES TIMEOUT 30)
add_test(NAME identity.create COMMAND voelund_identity_probe create "${CMAKE_CURRENT_BINARY_DIR}/fixtures/identity.volund")
set_tests_properties(identity.create PROPERTIES FIXTURES_SETUP identity_file)
add_test(NAME identity.read_edit COMMAND voelund_identity_probe read "${CMAKE_CURRENT_BINARY_DIR}/fixtures/identity.volund" "${CMAKE_CURRENT_BINARY_DIR}/fixtures/identity-rewritten.volund")
set_tests_properties(identity.read_edit PROPERTIES FIXTURES_REQUIRED identity_file FIXTURES_SETUP identity_rewritten)
add_test(NAME identity.reread COMMAND voelund_identity_probe read "${CMAKE_CURRENT_BINARY_DIR}/fixtures/identity-rewritten.volund")
set_tests_properties(identity.reread PROPERTIES FIXTURES_REQUIRED identity_rewritten)
add_test(NAME identity.negative COMMAND voelund_identity_probe negative "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
set_tests_properties(identity.create identity.read_edit identity.reread identity.negative PROPERTIES TIMEOUT 30)
foreach(toolkit IN ITEMS TKBin TKCAF TKPrim)
    get_target_property(dependencies ${toolkit} INTERFACE_LINK_LIBRARIES)
    message(STATUS "D08 ${toolkit} transitive modules: ${dependencies}")
endforeach()
